# Python dependencies for the mt5api HTTP server that runs inside the Windows VM.
#
# Tracked in git ON PURPOSE. This replaces the retired, gitignored
# config/requirements.txt that Dockerfile.test still pointed at long after it
# was removed, which broke `make test` with a Docker COPY error.
#
# scripts/start.bat installs this same set inline on every boot.
# tests/test_requirements_sync.py asserts the two lists stay identical, so the
# next person to add a dependency can't let them drift apart again.
#
# Dockerfile.test strips MetaTrader5 before installing — that wheel is
# Windows-only. tests/conftest.py injects a stub for it at import time.
#
# Every package is pinned to an exact version (.agents/rules/security.md, SEC14),
# the versions a running deployment was live-tested with. Bump one only to a
# release that has been on PyPI for at least seven days, and run the live suite.
#
# numpy 1.x: MetaTrader5 wheels are built against numpy 1.x and break silently
# under numpy 2.x. Read calls (account_info, copy_rates, ...) keep working, but
# order_send fails immediately with (-2, 'Unnamed arguments not allowed'). Move
# to numpy 2 only once MetaQuotes ships a wheel rebuilt for it.
pyyaml==6.0.3
MetaTrader5==5.0.5640
numpy==1.26.4
flask==3.1.3
waitress==3.0.2
flask-compress==1.24
psutil==7.2.2
# MCP interface (mt5api/mcp_server.py) mounted at /mcp. a2wsgi bridges the ASGI
# MCP app into the Flask/WSGI stack served by waitress (see mt5api/main.py).
# v2 removed mcp.server.fastmcp, which both MCP surfaces import.
mcp==1.28.0
a2wsgi==1.10.10
